<?php

namespace Drupal\yp_memorial\Controller;

use Drupal\Core\Controller\ControllerBase;
use Drupal\webform\Entity\WebformSubmission;

class MemoriesController extends ControllerBase {

  public function page() {
    $output = '<div class="yp-memories-wrap">';
    $output .= '<div class="yp-memory-submit"><a href="/webform/yp_memories">Share a Memory or Photo</a></div>';

    $sids = \Drupal::entityQuery('webform_submission')
      ->condition('webform_id', 'yp_memories')
      ->sort('created', 'DESC')
      ->accessCheck(FALSE)
      ->execute();

    if ($sids) {
      $submissions = WebformSubmission::loadMultiple($sids);

      foreach ($submissions as $submission) {
        $data = $submission->getData();

        $name = !empty($data['name']) ? htmlspecialchars($data['name']) : 'Anonymous';
        $memory = !empty($data['memory']) ? nl2br(htmlspecialchars($data['memory'])) : '';
        $created = date('F j, Y', $submission->getCreatedTime());

        $photo_html = '';

        if (!empty($data['photo'])) {
          $fid = is_array($data['photo']) ? reset($data['photo']) : $data['photo'];
          $file = \Drupal\file\Entity\File::load($fid);

          if ($file) {
            $url = \Drupal::service('file_url_generator')->generateAbsoluteString($file->getFileUri());
            $photo_html = '<div class="yp-memory-photo"><img src="' . $url . '" alt=""></div>';
          }
        }

        $output .= '<article class="yp-memory-card">';
        $output .= $photo_html;
        $output .= '<div class="yp-memory-name">' . $name . '</div>';
        $output .= '<div class="yp-memory-date">' . $created . '</div>';
        $output .= '<div class="yp-memory-text">' . $memory . '</div>';
        $output .= '</article>';
      }
    }
    else {
      $output .= '<div class="yp-memory-empty">No memories have been shared yet.</div>';
    }

    $output .= '</div>';

    return [
      '#markup' => $output,
    ];
  }
