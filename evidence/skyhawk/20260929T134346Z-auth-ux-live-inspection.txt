+/**
+ * Rewords Drupal core's generic login failure without changing its semantics.
+ */
+function skyhawk_site_fixes_user_login_message_validate(
+  array &$form,
+  \Drupal\Core\Form\FormStateInterface $form_state
+): void {
+  $errors = $form_state->getErrors();
+
+  if (!$errors) {
+    return;
+  }
+
+  $changed = FALSE;
+
+  foreach ($errors as $name => $message) {
+    $plain = html_entity_decode(
+      strip_tags((string) $message),
+      ENT_QUOTES | ENT_HTML5,
+      'UTF-8'
+    );
+
+    if (
+      stripos(
+        $plain,
+        'Unrecognized username or password'
+      ) !== FALSE
+    ) {
+      $errors[$name] = t(
+        'Paddles sent you around. Watch the ball next time around! We couldn\'t sign you in with those credentials. Check the creds or use <a href=":url">Reset your password</a> below.',
+        [
+          ':url' => \Drupal\Core\Url::fromRoute('user.pass')->toString(),
+        ]
+      );
+
+      $changed = TRUE;
+    }
+  }
+
+  if (!$changed) {
+    return;
+  }
+
+  $form_state->clearErrors();
+
+  foreach ($errors as $name => $message) {
+    $form_state->setErrorByName($name, $message);
+  }
 }
