

/**
 * Implements hook_node_access().
 *
 * SKYHAWK_READYROOM_ACCESS_V1 (Gene decision 2026-10-13): Ready Room pages
 * (article_readyroom) are members-only by default. Viewing requires the
 * authorized_user or admin role (same rule as SA Journal PDFs). Pages whose
 * node IDs are listed in state 'skyhawk_readyroom_public_nids' stay public.
 * Replaces the earlier menu-only hiding, which did not restrict access.
 */
function skyhawk_site_fixes_node_access(\Drupal\node\NodeInterface $node, $op, \Drupal\Core\Session\AccountInterface $account) {
  if ($op !== 'view' || $node->bundle() !== 'article_readyroom') {
    return \Drupal\Core\Access\AccessResult::neutral();
  }
  $public = \Drupal::state()->get('skyhawk_readyroom_public_nids', []);
  $is_public = in_array((int) $node->id(), array_map('intval', $public), TRUE);
  $member = $account->hasPermission('bypass node access')
    || in_array('authorized_user', $account->getRoles(), TRUE)
    || in_array('admin', $account->getRoles(), TRUE);
  $result = ($is_public || $member)
    ? \Drupal\Core\Access\AccessResult::neutral()
    : \Drupal\Core\Access\AccessResult::forbidden('Ready Room pages are for members.');
  return $result->addCacheContexts(['user.roles', 'user.permissions'])
    ->addCacheableDependency($node)
    ->addCacheTags(['skyhawk_readyroom_public']);
}
